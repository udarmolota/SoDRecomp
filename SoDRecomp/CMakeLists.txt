cmake_minimum_required(VERSION 3.20)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# No C++20 modules here; skip CMake's dependency scanning of every source.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(BUILD_SHARED_LIBS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

project(SoDRecomp-ALL C CXX)

if (NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "The recompiled code needs Clang (clang-cl on Windows).")
endif()

# Third-party tools and SDKs (xenia-canary's third_party, the upscaler SDKs) are DAORecomp's
# installer downloads, used read-only from its checkout (XDroid360's recomp submodule,
# ../cmake/XDroid360Root.cmake); XenonRecomp is SoD's patched copy
# (../XenonRecomp, patched by scripts/patch_xenonrecomp.py).
include(${CMAKE_SOURCE_DIR}/../cmake/XDroid360Root.cmake)
set(SOD_TOOLS_ROOT "${XDROID360_ROOT}/recomp/tools" CACHE PATH "DAORecomp's tools folder (read-only)")
set(XENON_ROOT ${CMAKE_SOURCE_DIR}/../XenonRecomp CACHE PATH "SoD's patched XenonRecomp")
# Xenia's third-party libraries (xenia-canary checkout with its submodules).
set(SOD_XENIA_THIRD_PARTY ${SOD_TOOLS_ROOT}/xenia-canary/third_party)
# The recompiled game code (generated by XenonRecomp, never committed).
set(SOD_PPC_DIR ${CMAKE_SOURCE_DIR}/../ppc CACHE PATH "XenonRecomp output directory (phase 1)")
# Title Update 6's recompiled code (optional: built in when ppc_tu6/ was generated;
# needed by Awakening and the DLC campaigns).
set(SOD_PPC_TU6_DIR ${CMAKE_SOURCE_DIR}/ppc_tu6 CACHE PATH "XenonRecomp output directory for Title Update 6")

# The processor the build is for. The installer picks x86-64-v3 or sandybridge for the
# game from the PC's own (installer/pipeline.py target_cpu); this default (any AVX
# processor, UnleashedRecomp's baseline) is for direct builds and the downloaded
# launcher. The exe checks for it when it starts (SoDRecomp/cpu_check.cpp).
set(SOD_MARCH "sandybridge" CACHE STRING "Target CPU (-march): sandybridge or x86-64-v3")
add_compile_options(-march=${SOD_MARCH})
# No fused multiply-adds where the code has none (the same float results on
# every target).
add_compile_options(/clang:-ffp-contract=off)

# Xenia's GPU/UI/base: (SoD) the copy shared with DAO, DAORecomp's vendored and patched Xenia,
# used read-only from its checkout (SOD_SHARED_XENIA; the shared plan's A5-xv). It takes
# DAO_XENIA_THIRD_PARTY from here, and finds the upscaler SDKs and thirdparty/ through its own
# DAO_RECOMP_ROOT (DAORecomp's root by default). Comes first so its fmt and xxhash targets are
# the ones XenonUtils picks up.
set(SOD_SHARED_XENIA "${XDROID360_ROOT}/recomp/xenia" CACHE PATH "DAORecomp's Xenia (read-only)")
set(DAO_XENIA_THIRD_PARTY ${SOD_XENIA_THIRD_PARTY})
add_subdirectory(${SOD_SHARED_XENIA} ${CMAKE_BINARY_DIR}/xenia)

# SDL3 (static): gamepads incl. DualSense, haptics, audio output, and the
# launcher window (2D renderer, Direct3D 11/12 backends only).
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_RENDER ON CACHE BOOL "" FORCE)
set(SDL_RENDER_VULKAN OFF CACHE BOOL "" FORCE)
set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
add_subdirectory(thirdparty/SDL EXCLUDE_FROM_ALL)

# XenonUtils (XEX structures, byteswap helpers).
add_subdirectory(${XENON_ROOT} ${CMAKE_BINARY_DIR}/XenonRecomp EXCLUDE_FROM_ALL)

# What players download: SoDRecomp.exe with the launcher and installer only. It has no
# game code (no ppc/): installing recompiles the player's game into the full one.
option(SOD_LAUNCHER_ONLY "Build the launcher and installer without the game" OFF)
if (SOD_LAUNCHER_ONLY)
    add_subdirectory(SoDRecomp)
    return()
endif()

add_subdirectory(SoDRecompLib)
add_subdirectory(SoDRecomp)
